Users/portfolio Folder & Express.js Server

To host this portfolio inside your Users/<YourUsername>/portfolio directory using Express.js, run the commands below or copy the ready-made server.js:

// 1. Folder Structure inside your Users folder:
~/portfolio/
├── package.json
├── server.js
└── public/
└── index.html <-- (Save this file here)
// 2. server.js (Express.js Single-Page Portfolio Server) 
const express = require('express');
const path = require('path');
const app = express();
const PORT = process.env.PORT || 3000;

// Middleware for JSON & serving static assets from /public
app.use(express.json());
app.use(express.static(path.join(__dirname, 'public')));

// Single-page portfolio entry route
app.get('/', (req, res) => {
    res.sendFile(path.join(__dirname, 'public', 'index.html'));
});

// Optional API endpoint for the portfolio Contact Ping form
app.post('/api/ping', (req, res) => {
    const { name, message } = req.body;
    console.log(`[Portfolio Ping] From: ${name} | Msg: ${message}`);
    res.json({ status: 'success', receivedAt: new Date().toISOString() });
});

app.listen(PORT, () => {
    console.log(`🚀 Portfolio running at http://localhost:${PORT}`);
});
// 3. Run in Terminal:
mkdir -p ~/portfolio/public && cd ~/portfolio && npm init -y && npm install express && node server.js
